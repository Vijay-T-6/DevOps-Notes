name = "hello world"
i = 0
result = []

while i < len(name):
    char = name[i]
    
    # Check if character is already counted
    j = 0
    found = False
    while j < len(result):
        if result[j][0] == char:
            found = True
            break
        j += 1

    # If not counted, count its frequency
    if not found:
        count = 0
        k = 0
        while k < len(name):
            if name[k] == char:
                count += 1
            k += 1
        result += [(char, count)]  # manually add tuple to result list

    i += 1

# Print the result
j = 0
while j < len(result):
    print(result[j][0], ":", result[j][1])
    j += 1
