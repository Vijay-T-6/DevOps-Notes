how to convert lower case file names to a uppercase filenames

for file in *; do
  upper=$(echo "$file" | tr '[:lower:]' '[:upper:]')
  if [ "$file" != "$upper" ]; then
    mv "$file" "$upper"
  fi
done

write a script to check whether the website is up are down

import requests

def check_website(url):
    try:
        response = requests.get(url, timeout=5)
        if response.status_code == 200:
            print(f"{url} is UP ✅")
        else:
            print(f"{url} is DOWN ❌ (Status Code: {response.status_code})")
    except requests.exceptions.RequestException as e:
        print(f"{url} is DOWN ❌ (Error: {e})")

# Example usage
check_website("https://www.google.com")
check_website("https://www.nonexistentwebsite12345.com")
